#version 330 core

layout (location = 0) in vec3  aPos;
layout (location = 1) in vec3  aNormal;
layout (location = 2) in float aIntersects;
uniform mat4 trans;
uniform mat4 view;
uniform mat4 projection;

out vec3 Normal;
out vec3 FragPos;
out vec3 Color;
out float Intersects;

void main() {
   gl_Position = projection * view * trans * vec4(aPos, 1.0);
   FragPos = vec3(trans * vec4(aPos, 1.0));
   Normal = mat3(transpose(inverse(trans))) * aNormal; 
   Intersects = aIntersects;
}
